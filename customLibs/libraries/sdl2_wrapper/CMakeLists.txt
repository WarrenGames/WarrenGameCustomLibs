cmake_minimum_required(VERSION 3.13)

project( sdl2_wrapper LANGUAGES CXX)

file( GLOB_RECURSE HEADERS CONFIGURE_DEPENDS "sources/*.h" )

file( GLOB_RECURSE SOURCES CONFIGURE_DEPENDS "sources/*.cpp" )

add_library(
	${PROJECT_NAME} 
	STATIC
	${SOURCES} 
	${HEADERS} 
	${INC_DIRS}
)

source_group( TREE ${CMAKE_CURRENT_SOURCE_DIR}
	FILES ${HEADERS} ${SOURCES}
)

target_include_directories(
	${PROJECT_NAME}
	PRIVATE ${SDL2_INCLUDE_DIRS}
	PRIVATE ${SDL2_IMAGE_INCLUDE_DIRS}
	PRIVATE ${SDL2_TTF_INCLUDE_DIRS}
	PRIVATE ${SDL2_MIXER_INCLUDE_DIRS}
	PRIVATE sources
	PRIVATE ../genericLib/sources
)

target_link_libraries(	
	${PROJECT_NAME} 
	PUBLIC ${SDL2_LIBRARIES} 
	PUBLIC ${SDL2_IMAGE_LIBRARIES} 
	PUBLIC ${SDL2_TTF_LIBRARIES} 
	PUBLIC ${SDL2_MIXER_LIBRARIES}
	PRIVATE genericLib
)

install( DIRECTORY sources/ DESTINATION includes/sdl2_wrapper FILES_MATCHING PATTERN "*.h" )

install(
	TARGETS ${PROJECT_NAME}
	LIBRARY DESTINATION ${BuiltSharedLibsDir}
	ARCHIVE DESTINATION ${BuiltStaticLibsDir}
)
